cmake_minimum_required(VERSION 3.22)
project(video2_whisper LANGUAGES C CXX)
include(../cpu-only.cmake)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
if(VIDEO2_WHISPER_FP16 AND NOT ANDROID)
    message(FATAL_ERROR "The hardware-checked FP16 variant is Android-only")
endif()
if(VIDEO2_WHISPER_FP16)
    # Compile only the ggml CPU backend for the FP16 extension. The generic
    # sibling probes Linux HWCAP before ART can load this library.
    set(GGML_CPU_ARM_ARCH "armv8-a+fp16" CACHE STRING "" FORCE)
endif()
add_subdirectory("${VIDEO2_SOURCES}" vendor EXCLUDE_FROM_ALL)
if(VIDEO2_WHISPER_FP16)
    video2_jni(video2_whisper_fp16 whisper)
    target_compile_definitions(video2_whisper_fp16 PRIVATE VIDEO2_WHISPER_FP16)
else()
    video2_jni(video2_whisper whisper)
endif()
